import os

import numpy as np
import tensorflow as tf
from flask import Flask, jsonify, request
from tensorflow import keras
from tensorflow.keras.applications.efficientnet import preprocess_input
from tensorflow.keras.preprocessing import image

app = Flask(__name__)
MODEL_PATH = '../model/best_weights.keras'

# Opsi loading model
try:
    # Cara 1: Load model standar dengan compile=False
    print("Mencoba loading model cara 1...")
    model = keras.models.load_model(MODEL_PATH, compile=False)
    print("Berhasil load model!")
except Exception as e:
    print(f"Error pada percobaan pertama: {e}")
    try:
        # Cara 2: Load dengan custom objects untuk BatchNormalization
        print("Mencoba loading model cara 2...")
        model = keras.models.load_model(
            MODEL_PATH,
            custom_objects={
                'BatchNormalization': keras.layers.BatchNormalization
            },
            compile=False
        )
        print("Berhasil load model!")
    except Exception as e:
        print(f"Error pada percobaan kedua: {e}")
        try:
            # Cara 3: Gunakan tf.keras.saving.load_model
            print("Mencoba loading model cara 3...")
            model = tf.keras.saving.load_model(MODEL_PATH, compile=False)
            print("Berhasil load model!")
        except Exception as e:
            print(f"Semua percobaan load model gagal: {e}")
            raise Exception("Tidak dapat memuat model. Periksa format dan versi model.")

# Label kelas (sesuaikan dengan model kamu)
CLASS_NAMES = ['First_Degree', 'Second_Degree', 'Third_Degree']

# Ukuran gambar yang dipakai oleh model
IMG_SIZE = (224, 224)

@app.route('/')
def home():
    return 'Skin Burn Prediction API is running!'

@app.route('/predict', methods=['POST'])
def predict():
    if 'image' not in request.files:
        return jsonify({'error': 'No image uploaded'}), 400

    file = request.files['image']
    if file.filename == '':
        return jsonify({'error': 'Empty filename'}), 400

    try:
        # Load dan preprocess gambar
        # Simpan file sementara untuk dibaca sebagai gambar
        temp_path = "temp_image.jpg"
        file.save(temp_path)
        
        img = image.load_img(temp_path, target_size=IMG_SIZE)
        img_array = image.img_to_array(img)
        img_array = preprocess_input(img_array)
        img_array = np.expand_dims(img_array, axis=0)  # Tambah batch dimensi

        # Prediksi
        predictions = model.predict(img_array)
        predicted_class = CLASS_NAMES[np.argmax(predictions[0])]
        confidence = float(np.max(predictions[0]))

        # Hapus file temporary
        if os.path.exists(temp_path):
            os.remove(temp_path)

        return jsonify({
            'prediction': predicted_class,
            'confidence': round(confidence * 100, 2)
        })
    except Exception as e:
        return jsonify({'error': str(e)}), 500

if __name__ == '__main__':
    app.run(debug=True)
